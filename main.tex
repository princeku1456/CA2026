% =========================================
\usepackage{titlesec}

% Style table of contents in blue (text only, not page numbers)

\renewcommand{\cftchapfont}{\bfseries\color{bluetext}}
\renewcommand{\cftsecfont}{\color{bluetext}}

\setlength{\cftbeforesecskip}{2pt}

% Chapter formatting - clean and simple
\titleformat{\chapter}[display]
  {\bfseries\Large}
  {\chaptername\ \thechapter}
  {0.5em}
  {\titlerule\vspace{0.5em}}
  [\vspace{0.5em}\titlerule]

% Section formatting
\titleformat{\section}
  {\bfseries\large}
  {\thesection}
  {0.8em}
  {}

% Subsection formatting
\titleformat{\subsection}
  {\bfseries\normalsize}
  {\thesubsection}
  {0.6em}
  {}

% Subsubsection formatting
\titleformat{\subsubsection}
  {\itshape\normalsize}
  {\thesubsubsection}
  {0.5em}
  {}

% Spacing adjustments
\titlespacing*{\chapter}{0pt}{-10pt}{15pt}

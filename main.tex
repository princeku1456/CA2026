% ── CHAPTER: badge-style panel ──
% Double gold rule wrapping a display line;
% chapter number sits in small-caps above the title.
\titleformat{\chapter}[display]
  {\bfseries\Large\color{darktext}}
  {%
    {\color{goldrule}\titlerule[1.2pt]}%
    \vspace{4pt}%
    {\normalfont\small\scshape\color{goldrule}\chaptername\ \thechapter}%
  }
  {2pt}
  {\vspace{2pt}}
  [{\vspace{6pt}\color{goldrule}\titlerule[1.2pt]}]

% ── SECTION: crimson, with left gold bar via \llap ──
\titleformat{\section}
  {\headingfont\bfseries\large\color{sectioncolor}}
  {%
    \llap{{\color{goldrule}\rule[-2pt]{2.5pt}{11pt}}\hspace{5pt}}%
    \thesection
  }
  {0.7em}
  {}

% ── SUBSECTION: imperial violet ──

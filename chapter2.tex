\section{Pillar 5: Infrastructure}
\subsection{The Great Nicobar Project: Strategic, Economic and Ecological Framework}

\begin{itemize}
    \item \textbf{Context \& Strategic Significance:} 
    \begin{enumerate}
        \item Aimed at transforming Great Nicobar into a strategic maritime and economic hub in the Indian Ocean Region, strengthening India's presence in the Andaman Sea and Southeast Asia.
        \item Seeks to capture transshipment trade currently lost to foreign ports like Colombo, Singapore, and Klang due to India's lack of deep-water berths.
    \end{enumerate}

    \item \textbf{Core Infrastructure Components:}
    \begin{enumerate}
        \item \textbf{International Container Transhipment Terminal (ICTT):} Located at Galathea Bay with a capacity of 14.2 million Twenty-Foot Equivalent Units (TEU).
        \item \textbf{Greenfield International Airport:} Designed to handle 4,000 Peak Hour Passengers (PHP), projecting an initial 1 million passengers annually, scaling up to 10 million.
        \item \textbf{Township and Area Development:} A planned urban center spanning 16,610 hectares to support port-led residential, commercial, and institutional requirements.
        \item \textbf{Power Plant:} A 450 MVA hybrid gas and solar-based power plant designed with N-1 contingency (redundancy) to phase out diesel generators and reduce emissions.
    \end{enumerate}

    \item \textbf{Forest Diversion \& Compensatory Afforestation:}
    \begin{enumerate}
        \item Diverts 1.82\% of the total forest cover of the A\&N Islands.
        \item Preservation of 65.99 sq. km of land as strict green zones within the project area.
    \end{enumerate}

    \item \textbf{Tribal Welfare \& Rights Protection:}
    \begin{enumerate}
        \item Inhabited by Particular Vulnerable Tribal Groups (PVTGs): Shompens ($\approx$237, hunter-gatherers) and Nicobarese ($\approx$1,094, coastal fishing communities).
        \item \textbf{No Displacement/Relocation:} Project aligns with the Shompen Policy (2015), Jarawa Policy (2004), and Article 338A(9) of the Constitution. No tribal habitations are being relocated.
        \item \textbf{Tribal Reserve Land Balance:} 84.10 sq. km overlaps with the current Tribal Reserve. Within this, 11.032 sq. km is already utilized as revenue land since 1972. Thus, 73.07 sq. km is being de-notified for the project. To balance this, 76.98 sq. km is being re-notified as Tribal Reserve, creating a \textbf{net addition of 3.912 sq. km} to the reserve area.
    \end{enumerate}
\end{itemize}

\subsection{India's Health Transformation}

\subsubsection*{Overview}
\begin{itemize}
  \item NCDs: 60\% of all deaths; communicable disease burden declining
  \item Medical colleges doubled since 2014; 12 new AIIMS functional
\end{itemize}

\subsubsection*{Ayushman Bharat: Four Pillars}

\textbf{Pillar 1 — AB-PMJAY (Health Insurance)}
\begin{itemize}
  \item Coverage: ₹5 lakh/family/year $\rightarrow$ bottom 40\% ($\approx$12 cr families)
  \item Oct 2024 expansion: \textbf{Ayushman Bharat Vay Vandana} — all senior citizens $>$70 yrs
  \item \textbf{Key stats (as on June 5, 2026):}
    \begin{itemize}
      \item 44.14 cr Ayushman Cards created
      \item Out of pocket Expenditure reduced from 60\% to 32\%
      \item 40\% rural person have health insurance protection 
    \end{itemize}
  \item Ayushman App: 19 regional languages; eligibility check, e-card, hospital locator
\end{itemize}

\textbf{Pillar 2 — Ayushman Arogya Mandirs (AAM) / Primary Care}
\begin{itemize}
  \item 12 free services: NCD screening, oral/eye/ENT, mental health, teleconsultation, emergency care, free medicines \& diagnostics
  \item Cumulative footfall: $>$540 cr
  \item \textbf{Infrastructure (June 5, 2026):} 1.86+ lakh AAMs operational
\end{itemize}

\textbf{Pillar 3 — PM-ABHIM (Pandemic Preparedness)}
\begin{itemize}
  \item Launched: Oct 25, 2021; Outlay: ₹64,180 cr (FY2021--22 to 2025--26)
  \item Targets:
    \begin{itemize}
      \item 23,224 rural HWCs (10 high-focus states); 13,736 urban HWCs (all states)
      \item 3,389 block public health units (11 states); 744 integrated public health labs (all districts)
      \item 631 critical care blocks (districts $>$5 lakh population)
      \item 50 points of entry: bio-security + outbreak response; 12 central institutions: CCBs (150 beds each)
    \end{itemize}
\end{itemize}

\textbf{Pillar 4 — ABDM (Digital Health Ecosystem)}
\begin{itemize}
  \item Launched: Sep 2021; Core: \textbf{ABHA} (14-digit health ID, consent-based portable records)
  \item QR-based appointment registration; evidence-based population-level intervention
\end{itemize}

\subsubsection*{National Health Mission (NHM)}

\textbf{Maternal \& Child Health}
\begin{itemize}
  \item \textbf{PMSMA (Pradhan Mantri Surakshit Matritva Abhiyan)} (2016): Free antenatal check-ups; 7.47 cr pregnant women examined; 22,349 facilities
  \item \textbf{JSY}: Facility-based delivery incentives for BPL/SC/ST women
  \item \textbf{JSSK (Janani Shishu Suraksha Karyakram) }: Free delivery, drugs, diagnostics, diet, transport at public facilities (extended 2014: antenatal \& postnatal complications)
  \item \textit{Surakshit Matritva Aashwasan}: Guaranteed free dignified care at public facilities
  \item \textit{Stillbirth Surveillance}: Target $<$10/1,000 births by 2030
  \item \textit{HBNC (Home-Based New-born Care)}: 6 ASHA home visits post-birth; \textit{HBCYC}: 5 visits (3--15 months)
  \item \textit{RBSK (Rashtriya Bal Swasthya Karyakram)}: Screening for birth defects, delays, deficiencies
\end{itemize}

\textbf{Nutrition \& Adolescent Health}
\begin{itemize}
  \item \textit{Anaemia Mukt Bharat}: IFA supplementation for pregnant women, children, adolescents
  \item \textit{WIFS (Weekly Iron and Folic Acid Supplementation)}: Weekly IFA for adolescents; \textit{MAA (Mother's Absolute Affection)}: Breastfeeding promotion
  \item \textit{Menstrual Hygiene Scheme}: Information + sanitary products + safe disposal for adolescent girls
\end{itemize}

\textbf{Immunisation}
\begin{itemize}
  \item \textbf{UIP}: 2.67 cr newborns + 2.9 cr pregnant women/year; free vaccines against 12 diseases
  \item \textbf{Mission Indradhanush} (2014): Catch-up vaccination $\rightarrow$ 5.46 cr children + 1.32 cr pregnant women immunised
  \item NHFS 2023--24: $>$95\% children (12--23 months) vaccinated at public facility
  \item Zero-dose children: 0.11\% (2023) $\rightarrow$ 0.06\% (2024)
  \item WHO certified: Maternal \& Neonatal Tetanus eliminated (May 2015)
  \item \textbf{U-WIN} (2024 pan-India rollout): Digital vaccination tracking; 11.87 cr children + 3.96 cr pregnant women registered (Mar 2026)
  \item Feb 2026: Free HPV vaccine campaign (girls aged 14; $\sim$1.15 cr beneficiaries)
\end{itemize}

\subsubsection*{Affordable Medicines \& Diagnostics}

\begin{itemize}
  \item \textbf{PMBJP / Jan Aushadhi Kendras}: 18,000+ JAKs; generic medicines 50--80\% below market price
  \item \textbf{AMRIT Pharmacies} (2015): 50--90\% discount; 6.85 cr patients; 255+ outlets; ₹8,400 cr savings
  \item \textbf{Free Essential Diagnostics Initiative} (2015): 9/19/39/57 free tests at SC/PHC/CHC/DH levels; specialities: haematology, serology, biochemistry, pathology, microbiology, radiology, cardiology
  \item \textbf{Emergency Transport}: Dial 108 (critical/trauma) + Dial 102 (maternal/child); 3,044 ALS + 15,283 BLS ambulances + 3,918 PTVs; 19 boats + 81 bikes (hard-to-reach areas)
\end{itemize}

\subsubsection*{Digital \& Last-Mile Health Services}

\begin{itemize}
  \item \textbf{eSanjeevani} (Nov 2019): National telemedicine; 47 cr+ consultations; 2.34 lakh+ providers onboarded; \textbf{CDSS (Apr 2023)}: AI-enabled clinical alerts; 28.2 cr consultations benefited (till Nov 2025)
  \item \textbf{Tele-MANAS} (Oct 2022): Mental health; 20 languages; 36 states/UTs; 53 cells + 23 mentoring institutes; mobile app launched Oct 2024 (visually impaired accessible)
  \item \textbf{i-DRONE} (ICMR, 2021): 22,000 medicines; 7,700 km; 65 healthcare centres
\end{itemize}

\subsubsection*{AI in Healthcare}

\begin{itemize}
  \item \textbf{`Cough Against TB'}: Cough-analysis AI; 12--16\% additional TB detection; 1.62 lakh+ screened (Mar 2023--Nov 2025)
  \item \textbf{MadhuNetrAI}: Automated diabetic retinopathy screening; 38 facilities, 11 states; 14,000+ retinal images; 7,100 patients (Dec 2025)
\end{itemize}

\subsection{Rediscovery of Gynacantha khasiaca (Long-tailed Duskhawker)}

\begin{itemize}
    \item \textbf{Context \& Rediscovery:} 
    \begin{itemize}
        \item A rare dragonfly species, \textit{Gynacantha khasiaca} (commonly known as the long-tailed duskhawker), has resurfaced in the Changlang district of Arunachal Pradesh (\textbf{Namdapha National Park and Tiger Reserve}).
    \end{itemize}

    \item \textbf{About Dragonflies:} 
    \begin{enumerate}
        \item Possesses two large compound eyes, each comprised of thousands of tiny lenses and photoreceptor clusters.
        \item Features a near-$360^\circ$ field of vision and possesses the distinct behavioral trait of being able to hover perfectly still in mid-air.
        \item They serve as critical bioindicators and components of freshwater ecosystems, functioning simultaneously as apex invertebrate predators and vital prey within the aquatic food web.
        \item long-tailed duskhawker has also been historically documented across Assam, Meghalaya, West Bengal, Uttarakhand, and Maharashtra.
    \end{enumerate} 
\end{itemize}


\section{Miscellaneous}

\subsection{India's Green Pathway: Conservation to Climate Action}

\subsubsection*{Biodiversity Status}
\begin{itemize}
  \item One of 17 mega-biodiverse countries; 2.4\% land area $\rightarrow$ $\sim$8\% global species
  \item 96,000+ animal species; 47,000+ plant species; $\sim$50\% world's aquatic plant species
  \item Governance: Biological Diversity Act 2002 + National Biodiversity Action Plan + CBD (signatory 1992)
  \item \textbf{National Biodiversity Strategy and Action Plan  (NBSAP 2024--2030)} launched at CBD COP16, Cali: halt biodiversity loss by 2030; harmony with nature by 2050; 23 ministries involved; aligns with \textbf{Kunming-Montreal Global Biodiversity Framework}
\end{itemize}

\subsubsection*{Core Environmental Legislation}
\begin{itemize}
  \item Wildlife (Protection) Act 1972; Water Pollution Act 1974; Forest (Conservation) Act 1980
  \item Air Pollution Act 1981; Environment (Protection) Act 1986 (umbrella); Biological Diversity Act 2002
  \item CAQM Act 2021 -- air quality governance for NCR
\end{itemize}

\subsubsection*{Protected Areas \& Species Conservation}
\begin{itemize}
  \item Protected Areas: 745 (2014) $\rightarrow$ \textbf{1,134 (2025)}
  \item \textbf{Project Tiger}: Reserves 46 $\rightarrow$ \textbf{58}; newest -- Madhav TR (MP); tiger count \textbf{3,167 (2022)}
  \item \textbf{Project Elephant}: Reserves 26 $\rightarrow$ \textbf{33};
  \item \textbf{Project Cheetah}: Expanded to Gandhisagar WLS; planned -- Noradehi \& Banni Grasslands; total population \textbf{30} (incl. 19 India-born cubs, Dec 2025)
  \item \textbf{Project Snow Leopard}: 1st nationwide assessment (2019--23); population \textbf{718} -- Ladakh 477, Uttarakhand 124; SPAI 2.0 launched (Wildlife Week 2025)
  \item \textbf{Project Dolphin}: 2021--23 survey -- \textbf{6,327 riverine dolphins}; 2nd range-wide estimation launched Jan 2026 (Bijnor); covers Ganga, Indus, Brahmaputra, Sundarbans, Odisha
  \item \textbf{International Big Cat Alliance} (est. Apr 2023): 7 big cat species; Framework Agreement in force 23 Jan 2025; 18 member countries; led by India
  \item Wildlife Week 2025 (Oct 2--8): launched Project Dolphin Ph-II, Project Sloth Bear, Project Gharial, 4 national action plans; Centre of Excellence for Human-Wildlife Conflict; ``Tigers Outside Tiger Reserve'' project
\end{itemize}

\subsubsection*{Ecosystem Restoration}
\begin{itemize}
  \item \textbf{Biosphere Reserves}: 18 notified (91,425 km²); 13 under UNESCO WNBR; Cold Desert BR (HP) added Sep 2025
  \item \textbf{Ek Ped Maa Ke Naam}: 262.4 crore saplings planted (till Dec 2025)
  \item \textbf{MISHTI} (Mangrove Initiative for Shoreline Habitats \& Tangible Incomes): 4,536 ha restored (2025); 22,560 ha degraded mangrove identified across 13 States/UTs
  \item \textbf{Ramsar Sites}: 26 (2014) $\rightarrow$ \textbf{98 (Jan 2026)}; highest in Asia, 3rd globally; recent additions -- Patna Bird Sanctuary (UP), Chhari-Dhand (Gujarat); Udaipur \& Indore -- India's 1st Ramsar Wetland Cities
  \item \textbf{National Coastal Mission}: extended 2025--31; allocation \textbf{\rupee767 crore}; under NAPCC; integrated coastal zone management
  \item \textbf{Blue Flag Beaches}: 18 certified; 7 coastal states + 4 UTs; certified by Foundation for Environmental Education; implemented by MoEFCC under ICZM
\end{itemize}

\subsubsection*{Climate Policy Frameworks}
\begin{itemize}
  \item \textbf{NAPCC}: 9 National Missions (solar, energy efficiency, sustainable habitat, water, Himalayan ecosystems, Green India, sustainable agriculture, strategic knowledge + additional)
  \item \textbf{Updated NDC (2022)}: (i) 45\% reduction in GDP emissions intensity vs 2005 by 2030; (ii) 50\% installed power from non-fossil by 2030 -- both \textbf{achieved ahead of schedule} (36\% reduction and $>$50\% non-fossil by Jun 2025)
  \item \textbf{LT-LEDS} (2022): net-zero by \textbf{2070}
  \item \textbf{National Green Hydrogen Mission (2023)}: 5 MMT/yr production by 2030; 500 GW non-fossil capacity target by 2030
  \item \textbf{Mission LiFE}: 6 crore+ participants; 34 lakh+ events; 4.96 crore pledges (Dec 2025); 1 billion sustainable lifestyle adopters $\rightarrow$ $\sim$20\% global emission reduction potential
\end{itemize}

\subsubsection*{Renewable Energy \& Power Sector}
\begin{itemize}
  \item Total installed capacity (Jan 2026): \textbf{5,20,510.95 MW} -- Fossil: 2,48,541 MW; Non-fossil: \textbf{2,71,969 MW} (Nuclear 8,780 MW + RE 2,63,189 MW)
  \item RE share crossed \textbf{50\% of installed capacity} -- 5 yrs ahead of 2030 NDC target
  \item Global rank (IRENA 2025): Solar -- \textbf{3rd}; Wind -- \textbf{4th}; Total RE -- \textbf{4th}
  \item CO\textsubscript{2} intensity of power sector: 61.45 t/\rupee crore (2015-16) $\rightarrow$ \textbf{40.52 t/\rupee crore (2022-23)}
  \item Modhera (Gujarat): India's 1st 24$\times$7 solar-powered village (Oct 2022)
  \item Omkareshwar Floating Solar Park (MP): India's \& one of Asia's largest floating solar parks
\end{itemize}

\subsubsection*{Carbon Markets \& Industrial Transition}
\begin{itemize}
  \item \textbf{Carbon Credit Trading Scheme}: operationalised under Indian Carbon Market; compliance + offset mechanisms
  \item Jan 2026: GHG Emission Intensity targets notified for additional sectors; \textbf{490 obligated entities} across emission-intensive industries
  \item \textbf{CCUS}: Budget 2026--27 allocated \textbf{\rupee20,000 crore} over 5 years; covers chemicals + heavy industries
  \item Article 6 (Paris): National Designated Agency designated as national authority
  \item COP30 (Belém, Nov 2025): India stressed finance + technology transfer; joined Brazil's \textbf{Tropical Forests Forever Facility (TFFF)} as Observer
  \item \textbf{ISA}: 8th Assembly (Bharat Mandapam, Oct 2025); 550+ delegates; 30 ministerial reps
  \item \textbf{Montreal Protocol}: HCFC reduction -- \textbf{67.5\%} by 2025; Kigali Amendment (HFC phase-down) ratified 2021
\end{itemize}

\subsubsection*{Pollution Control \& Waste Management}
\begin{itemize}
  \item \textbf{NCAP} (Jan 2019): 130 cities; PM10 improvement -- 103/130 cities (2024-25 vs 2017-18 baseline); 64 cities $\geq$20\% reduction; 25 cities $\geq$40\% reduction
  \item \textbf{Fly Ash}: Generated 340+ MT (2024-25); utilised \textbf{332.63 MT} -- roads/flyovers 32\%, cement 27\%, bricks/tiles 14\%; 100\% utilisation mandated for thermal plants
  \item \textbf{Waste Recycling Plants}: 829 (2019-20) $\rightarrow$ \textbf{3,036 (2024-25)}
  \item \textbf{EPR}: notified for plastic packaging, e-waste, tyres, batteries; 71,401 producers + 4,447 recyclers registered; 375.11 lakh tonnes recycled; EPR certificates: 339.51 lakh tonnes generated
\end{itemize}
